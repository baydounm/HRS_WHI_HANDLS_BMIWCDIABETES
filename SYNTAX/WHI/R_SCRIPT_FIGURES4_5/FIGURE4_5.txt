# 4) Helper functions
parse_hr_ci <- function(x) {
  # Extracts HR, lower CI, upper CI from strings such as:
  # "1.51 (1.47, 1.56)" or "1.00 (Reference)"
  x <- as.character(x)
  hr  <- suppressWarnings(as.numeric(str_extract(x, "^[0-9.]+")))
  lcl <- suppressWarnings(as.numeric(str_match(x, "\\(([-0-9.]+),\\s*([-0-9.]+)\\)")[, 2]))
  ucl <- suppressWarnings(as.numeric(str_match(x, "\\(([-0-9.]+),\\s*([-0-9.]+)\\)")[, 3]))
  is_ref <- str_detect(x, regex("reference", ignore_case = TRUE))
  lcl[is_ref] <- hr[is_ref]
  ucl[is_ref] <- hr[is_ref]
  tibble(HR = hr, LCL = lcl, UCL = ucl, is_reference = is_ref)
}

save_forest <- function(p, filename, width = 12, height = 8) {
  ggsave(filename, p, width = width, height = height, dpi = 300)
}

# ==========================================================
# TABLE 5: BMI category forest plot
# ==========================================================

t5_raw <- read_excel(xlsx_file, sheet = "Table5", col_names = FALSE)

# Row 2 contains BMI category headers; rows 3 onward contain data.
t5_colnames <- c("Subgroup", as.character(unlist(t5_raw[2, 2:5])))
t5 <- t5_raw[-c(1, 2), 1:5]
colnames(t5) <- t5_colnames

t5_long <- t5 %>%
  mutate(
    Subgroup = as.character(Subgroup),
    Section = case_when(
      str_detect(Subgroup, ":$") ~ str_remove(Subgroup, ":$"),
      TRUE ~ NA_character_
    )
  ) %>%
  tidyr::fill(Section, .direction = "down") %>%
  filter(!str_detect(Subgroup, ":$")) %>%
  mutate(
    Subgroup = str_squish(Subgroup),
    Section = if_else(Section == "Overall", "Overall", Section)
  ) %>%
  pivot_longer(
    cols = -c(Subgroup, Section),
    names_to = "BMI_category",
    values_to = "HR_CI"
  ) %>%
  bind_cols(parse_hr_ci(.$HR_CI)) %>%
  filter(!is.na(HR)) %>%
  mutate(
    BMI_category = str_squish(BMI_category),
    Section = str_squish(Section),
    Label = if_else(Section == "Overall", "Overall", paste0(Section, ": ", Subgroup)),
    Label = factor(Label, levels = rev(unique(Label))),
    BMI_category = factor(
      BMI_category,
      levels = c("Overweight", "Obesity-1", "Obesity-2", "Obesity-3 ", "Obesity-3")
    )
  )

t5_plot <- ggplot(t5_long, aes(x = HR, y = Label, color = BMI_category, shape = BMI_category)) +
  geom_vline(xintercept = 1, linetype = "dashed", linewidth = 0.7, color = "gray35") +
  geom_errorbarh(aes(xmin = LCL, xmax = UCL), height = 0.18, linewidth = 0.75, alpha = 0.9) +
  geom_point(size = 3.2) +
  scale_x_log10(
    breaks = c(1, 1.25, 1.5, 2, 3, 4, 5, 6),
    labels = label_number(accuracy = 0.01)
  ) +
  labs(
    x = "Hazard ratio, log scale",
    y = NULL,
    color = "BMI category",
    shape = "BMI category"
  ) +
  theme_minimal(base_size = 12) +
  theme(
    plot.title = element_text(face = "bold", size = 16),
    plot.subtitle = element_text(size = 11),
    legend.position = "bottom",
    panel.grid.minor = element_blank(),
    axis.text.y = element_text(size = 9)
  )

print(t5_plot)
save_forest(t5_plot, "Table5_BMI_ForestPlot.png", width = 12, height = 8)
save_forest(t5_plot, "Table5_BMI_ForestPlot.pdf", width = 12, height = 8)
