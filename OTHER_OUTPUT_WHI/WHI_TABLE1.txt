. program define post_table1_cont
  1.     args var labelname handle
  2.     regress `var' i.bmi5_base if _st == 1
  3.     testparm i.bmi5_base
  4.     local p = r(p)
  5.     local cells ""
  6.     forvalues b=1/5 {
  7.         summarize `var' if _st == 1 & bmi5_base == `b'
  8.         local mean = r(mean)
  9.         local sd = r(sd)
 10.         local cell : display %9.2f `mean' " ± " %9.2f `sd'
 11.         local cells `"`cells' ("`cell'")"'
 12.     }
 13.     post `handle' ("`labelname'") ("Mean ± SD") `cells' (`p')
 14. end

. program define post_table1_cat
  1.     args var labelname handle
  2.     tab `var' bmi5_base if _st == 1, chi2
  3.     local p = r(p)
  4.     levelsof `var' if _st == 1, local(levels)
  5.     local first = 1
  6.     foreach lev of local levels {
  7.         local vl : value label `var'
  8.         if "`vl'" != "" local levlab : label `vl' `lev'
  9.         else local levlab "`lev'"
 10.         local cells ""
 11.         forvalues b=1/5 {
 12.             count if _st == 1 & bmi5_base == `b'
 13.             local denom = r(N)
 14.             count if _st == 1 & bmi5_base == `b' & `var' == `lev'
 15.             local num = r(N)
 16.             local pct = 100*`num'/`denom'
 17.             local cell : display %12.0fc `num' " (" %5.2f `pct' ")"
 18.             local cells `"`cells' ("`cell'")"'
 19.         }
 20.         if `first' == 1 {
 21.             post `handle' ("`labelname'") ("`levlab'") `cells' (`p')
 22.             local first = 0
 23.         }
 24.         else {
 25.             post `handle' ("`labelname'") ("`levlab'") `cells' (.)
 26.         }
 27.     }
 28. end
