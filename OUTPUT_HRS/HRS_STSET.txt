. ********************************************************************************
. ** 10. TABLES 3-4: COX MODELS AND HARRELL'S C-STATISTICS
. ** Only unadjusted and fully adjusted models are exported.
. ********************************************************************************
. 
. capture program drop make_cox_table

. program define make_cox_table
  1.     syntax, EXPOSURE(name) SHEET(string)
  2. 
.     tempfile results
  3.     postfile handle str24 table str30 model str40 exposure str100 category ///
>         double HR LCL UCL P CSTAT N_EVENTS N_SUBJECTS using `results', replace
  4. 
.     foreach model in unadjusted full {
  5.         if "`model'" == "unadjusted" local covars ""
  6.         if "`model'" == "full"       local covars "$COV_FULL"
  7. 
.         quietly stcox ib1.`exposure' `covars'
  8.         estimates store `sheet'_`model'
  9.         quietly estat concordance
 10.         capture local cstat = r(C)
 11.         if _rc local cstat = .
 12.         local nevents = e(N_fail)
 13.         local nsubjects = e(N)
 14. 
.         levelsof `exposure' if e(sample), local(levels)
 15.         foreach lev of local levels {
 16.             if `lev' != 1 {
 17.                 capture quietly lincom `lev'.`exposure', hr
 18.                 if _rc == 0 {
 19.                     local vl : value label `exposure'
 20.                     if "`vl'" == "" local cat "`lev'"
 21.                     else local cat : label `vl' `lev'
 22.                     post handle ("`sheet'") ("`model'") ("`exposure'") ("`cat'") ///
>                         (r(estimate)) (r(lb)) (r(ub)) (r(p)) (`cstat') (`nevents') (`nsubjects')
 23.                 }
 24.             }
 25.         }
 26. 
.         di "=============================================================="
 27.         di "`sheet' | `model' | exposure = `exposure'"
 28.         estat concordance
 29.         capture noisily estat phtest, detail
 30.         estat ic
 31.     }
 32.     postclose handle
 33. 
.     preserve
 34.         use `results', clear
 35.         export excel using "$xlsx", sheet("`sheet'", replace) firstrow(variables)
 36.     restore
 37. end
