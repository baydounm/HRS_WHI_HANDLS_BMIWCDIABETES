. program define post_stratified_cox
  1.     syntax, EXPOSURE(name) SHEET(string)
  2. 
.     tempfile results
  3.     postfile handle str24 table str30 stratum str80 subgroup str100 category ///
>         double HR LCL UCL P CSTAT N_EVENTS N_SUBJECTS using `results', replace
  4. 
.     * Overall fully adjusted model.
.     quietly stcox ib1.`exposure' $COV_FULL
  5.     quietly estat concordance
  6.     capture local cstat = r(C)
  7.     if _rc local cstat = .
  8.     local nevents=e(N_fail)
  9.     local nsubjects=e(N)
 10.     levelsof `exposure' if e(sample), local(levels)
 11.     foreach lev of local levels {
 12.         if `lev' != 1 {
 13.             capture quietly lincom `lev'.`exposure', hr
 14.             if _rc==0 {
 15.                 local vl : value label `exposure'
 16.                 if "`vl'" == "" local cat "`lev'"
 17.                 else local cat : label `vl' `lev'
 18.                 post handle ("`sheet'") ("Overall") ("All women") ("`cat'") ///
>                     (r(estimate)) (r(lb)) (r(ub)) (r(p)) (`cstat') (`nevents') (`nsubjects')
 19.             }
 20.         }
 21.     }
 22. 
.     * Age strata, fully adjusted except no age-category term; continuous age retained.

 40.                     }
 41.                 }
 42.             }
 43.         }
 44.     }
 45.
